⚠ One-time Supabase setup required. Run the following SQL in your Supabase dashboard → SQL Editor, then refresh this page:
CREATE TABLE time_off (
  id uuid DEFAULT gen_random_uuid() PRIMARY KEY,
  created_at timestamptz DEFAULT now(),
  store_key text NOT NULL,
  employee_name text NOT NULL,
  start_date date NOT NULL,
  end_date date NOT NULL,
  type text NOT NULL DEFAULT 'PTO',
  notes text,
  status text NOT NULL DEFAULT 'approved'
);
ALTER TABLE time_off ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow all" ON time_off FOR ALL USING (true) WITH CHECK (true);
